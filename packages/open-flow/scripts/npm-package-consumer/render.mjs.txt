const server = createServer((request, response) => {
  requests.push(request.url)
  response.setHeader('Access-Control-Allow-Origin', '*')
  if (request.url.startsWith('/providers')) {
    response.setHeader('ETag', `"providers-${providerVersion}"`)
    if (request.headers['if-none-match'] === `"providers-${providerVersion}"`) { response.writeHead(304).end(); return }
    const origin = `http://127.0.0.1:${server.address().port}`
    const iconSprite = { version: String(providerVersion), pixelRatio: 2, iconSize: 48, bleed: 2, width: 104, height: 52,
      lightUrl: `${origin}/provider-sprite/${providerVersion}/light.svg`, darkUrl: `${origin}/provider-sprite/${providerVersion}/dark.svg` }
    response.setHeader('Content-Type', 'application/json')
    response.end(JSON.stringify({version: 1, providers: [
      ...['mail', 'github'].map(serviceId => ({serviceId, serviceName: serviceId, iconSprite, iconSpritePosition: {x: 54, y: 2}})),
      {serviceId: 'files', serviceName: 'Files', icon: `${origin}/provider-original.svg`},
    ]}))
  } else if (request.url.startsWith('/provider-sprite/')) {
    if (providerVersion === 3) { response.writeHead(503).end(); return }
    const color = request.url.includes('/dark.') ? '#23b7ed' : providerVersion === 1 ? '#e72bbd' : '#6ee126'
    response.setHeader('Content-Type', 'image/svg+xml')
    response.end(`<svg xmlns="http://www.w3.org/2000/svg" width="104" height="52"><path fill="red" d="M2 2h48v48H2z"/><path fill="${color}" d="M54 2h48v48H54z"/></svg>`)
  } else if (request.url.startsWith('/provider-original')) {
    response.setHeader('Content-Type', 'image/svg+xml')
    response.end('<svg xmlns="http://www.w3.org/2000/svg" width="48" height="48"><path fill="#1f4ddf" d="M0 0h48v48H0z"/></svg>')
  } else if (request.url.startsWith('/redirect')) {
    response.writeHead(302, { Location: request.url.replace('/redirect', '/image') })
    response.end()
  } else {
    response.writeHead(200, { 'Content-Type': 'image/svg+xml' })
    response.end(image)
  }
})
let callerPage
async function verify(name, file, options) {
  const started = performance.now()
  const result = await flowToPng(browser, file, options)
  const durationMs = Math.round(performance.now() - started)
  assert.ok(Buffer.isBuffer(result.buffer))
  assert.equal(result.buffer.subarray(0, 8).toString('hex'), '89504e470d0a1a0a')
  assert.equal(result.buffer.readUInt32BE(16), result.width)
  assert.equal(result.buffer.readUInt32BE(20), result.height)
  assert.ok(Math.max(result.width, result.height) <= 16384)
  assert.ok(result.width * result.height <= 32_000_000)
  if (options?.aspectRatio && options.aspectRatio !== 'auto') {
    const [width, height] = options.aspectRatio.split(':').map(Number)
    assert.equal(result.width * height, result.height * width)
  }
  assert.equal(browser.isConnected(), true)
  assert.equal(browser.contexts().length, 1)
  assert.equal(await callerPage.title(), 'Caller-owned page')
  results.push({ name, width: result.width, height: result.height, hasImagePlaceholders: result.hasImagePlaceholders, durationMs })
  if (artifacts) await writeFile(path.join(artifacts, `${name}.png`), result.buffer)
  return result
}
async function cornerAlpha(result) {
  return callerPage.evaluate(async (source) => {
    const image = new Image()
    image.src = source
    await image.decode()
    const canvas = document.createElement('canvas')
    canvas.width = image.naturalWidth
    canvas.height = image.naturalHeight
    try {
      const context = canvas.getContext('2d')
      context.drawImage(image, 0, 0)
      return [
        context.getImageData(0, 0, 1, 1).data[3],
        context.getImageData(canvas.width - 1, canvas.height - 1, 1, 1).data[3],
        context.getImageData(Math.floor(canvas.width / 2), Math.floor(canvas.height / 2), 1, 1).data[3],
      ]
    } finally {
      canvas.width = canvas.height = 0
    }
  }, 'data:image/png;base64,' + result.buffer.toString('base64'))
}
async function colorPixels(result, rgb) {
  return callerPage.evaluate(async ({source, rgb}) => {
    const image = new Image(); image.src = source; await image.decode()
    const canvas = document.createElement('canvas'); canvas.width = image.naturalWidth; canvas.height = image.naturalHeight
    try {
      const context = canvas.getContext('2d'); context.drawImage(image, 0, 0)
      const pixels = context.getImageData(0, 0, canvas.width, canvas.height).data
      let count = 0
      for (let i = 0; i < pixels.length; i += 4) if (pixels[i] === rgb[0] && pixels[i + 1] === rgb[1] && pixels[i + 2] === rgb[2] && pixels[i + 3] === 255) count++
      return count
    } finally { canvas.width = canvas.height = 0 }
  }, {source: 'data:image/png;base64,' + result.buffer.toString('base64'), rgb})
}
