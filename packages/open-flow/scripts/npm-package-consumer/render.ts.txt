import type { Browser } from 'playwright-core'
import { flowToPng, type FlowPngAspectRatio, type FlowPngOptions, type FlowPngResult, type WorkflowFile } from '@oomol-lab/open-flow/render'
import { ControlClient, type ControlCacheDelegate } from '@oomol-lab/open-flow/control-api'
import { ProviderIconCache, type CacheDelegate } from '@oomol-lab/open-flow/provider-icons'

export async function render(browser: Browser, workflow: WorkflowFile): Promise<FlowPngResult> {
  const aspectRatio: FlowPngAspectRatio = '4:3'
  const options: FlowPngOptions = { language: 'zh-CN', dark: true, frame: true, layout: 'compact', aspectRatio, title: '发布包验证', allowNetwork: false }
  const result = await flowToPng(browser, workflow, options)
  const bytes: Buffer = result.buffer
  const width: number = result.width
  const replaced: boolean = result.hasImagePlaceholders
  void [bytes, width, replaced]
  return result
}

export async function renderProviders(browser: Browser, workflow: WorkflowFile, delegate: ControlCacheDelegate): Promise<FlowPngResult> {
  const sharedStorage: CacheDelegate = delegate
  const client = new ControlClient(fetch, 'api', { cache: {scope: 'consumer', delegate, mode: 'offline'} })
  const iconCache = new ProviderIconCache({scope: 'consumer', delegate: sharedStorage, mode: 'offline'})
  await client.listConnectorProviders(undefined, undefined, 'en', {cacheOnly: true})
  return flowToPng(browser, workflow, {client, iconCache, allowNetwork: false})
}
